"""
Domain events for `domains/cooperatives`.

These are plain dataclasses only — no dispatch mechanism lives here. The Celery-enqueue /
event-to-task plumbing is shared infra (other domains will emit events too), imported from
there. A service emits one of these via `self.events.dispatch(...)` only AFTER its
UnitOfWork.commit() has succeeded — never before, and never if the commit failed.

Every event carries the ids needed to look the thing back up (never the full ORM object — by
the time notifications processes it async, that instance may be stale/detached) plus
`actor_id` where there's a clear acting user, so notifications can render "X changed by Y" and
can exclude the actor from the notified-members list if desired.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime, timezone
from uuid import UUID


def _utcnow() -> datetime:
    return datetime.now(timezone.utc)


@dataclass(frozen=True)
class DomainEvent:
    """Base class. `occurred_at` is stamped at construction time (event-creation time), not
    commit time — the two are close enough in practice and this keeps the dataclass simple.
    """

    occurred_at: datetime = field(default_factory=_utcnow, kw_only=True)


@dataclass(frozen=True)
class CooperativeActivated(DomainEvent):

from __future__ import annotations
from uuid import UUID


class CooperativeDomainError(Exception):
    """Base class for every exception raised out of this domain's service layer.
 
    Lets the API layer add a single blanket `except CooperativeDomainError` fallback (mapped to
    a generic 400/500) without having to enumerate every subclass, while still being able to
    special-case the ones below to their specific status codes.
    """



class CooperativeNotFoundError(CooperativeDomainError):
    """Raised by get_cooperative() / any lookup when the id doesn't exist. -> 404"""
 
    def __init__(self, cooperative_id: UUID) -> None:
        self.cooperative_id = cooperative_id
        super().__init__(f"Cooperative {cooperative_id} not found")
 

class GoalNotFoundError(CooperativeDomainError):
    """Raised by get_goal() / any lookup when the id doesn't exist. -> 404"""
 
    def __init__(self, goal_id: UUID) -> None:
        self.goal_id = goal_id
        super().__init__(f"Goal {goal_id} not found")
 
 
class CooperativeNotConfiguredError(CooperativeDomainError):
    """Raised by activate() when governance/membership/financial rules aren't all set. -> 409
 
    `missing` lists exactly which of ["governance_rules", "membership_rules",
    "financial_rules"] are absent — this is what the frontend "Setup incomplete: X, Y" banner
    reads, so keep it precise rather than collapsing to a generic message.
    """
 
    def __init__(self, missing: list[str]) -> None:
        self.missing = missing
        super().__init__(f"Cooperative not configured, missing: {', '.join(missing)}")
 
 
class IllegalStatusTransitionError(CooperativeDomainError):
    """Raised whenever a status-machine guard fails (wrong current state for the transition
    attempted, e.g. activate() called on a COMPLETED cooperative, or a membership/quorum guard
    inside activate() fails). -> 409
    """
 
    def __init__(self, from_status: str, to_status: str) -> None:
        self.from_status = from_status
        self.to_status = to_status
        super().__init__(f"Cannot transition from {from_status} to {to_status}")
 
 
class NotAuthorizedError(CooperativeDomainError):
    """Raised when acting_user_id lacks the admin/owner role required for the action. -> 403
 
    Deliberately carries no cooperative_id / role detail in the message — that's for the log
    line (which has full context via `extra={}`), not for what gets surfaced to the caller.
    """
 
 
class IncompleteGoalContractError(CooperativeDomainError):
    """Raised by create_goal_contract() when a beneficiary_pattern-required field is missing.
    -> 422
 
    `missing` lists exactly which required_fields from the pattern's contract_schema weren't
    supplied — same "precise list, not a generic 400" pattern as CooperativeNotConfiguredError.
    """
 
    def __init__(self, missing: list[str]) -> None:
        self.missing = missing
        super().__init__(f"Goal contract incomplete, missing: {', '.join(missing)}")
 
 
class InvalidGoalContractError(CooperativeDomainError):
    """Raised by create_goal_contract() for pattern-specific cross-field invariant violations
    (e.g. pooled_for_shared missing exit_rule_type when valuation_method is set; pooled_for_one
    beneficiary_user_id not an actual cooperative member). -> 422
    """
 
    def __init__(self, reason: str) -> None:
        self.reason = reason
        super().__init__(reason)
 
 
class OwnershipShareOverallocatedError(CooperativeDomainError):
    """Raised when a new/updated share_pct would push a goal's total shares over 100%, or when
    GoalService.activate() finds a pooled_for_shared goal whose shares don't sum to exactly
    100%. -> 422
    """
 
    def __init__(self, current_total_pct: float) -> None:
        self.current_total_pct = current_total_pct
        super().__init__(f"Ownership shares would total {current_total_pct}%, exceeding 100%")

# `domains/cooperatives`

Replaces `domains/circles`. Implements the Cooperative Engine from the
"Database Design: Goals + Governance Core" doc — governance/membership/
financial/decision rule modules, plus the beneficiary-pattern /
goal-category split.

## Where this goes in your tree

Drop this whole folder in as:

```
apps/backend/src/domains/cooperatives/
├── __init__.py
├── README.md
└── models/
    ├── __init__.py
    ├── enums.py
    ├── mixins.py        # Base, UUIDPrimaryKeyMixin, TimestampMixin, VersionedMixin
    ├── cooperative.py    # CoopCooperative
    ├── rules.py           # CoopGovernanceRule, CoopMembershipRule, CoopFinancialRule, CoopDecisionRule
    ├── goals.py            # CoopBeneficiaryPattern, CoopGoalCategory, CoopGoal, CoopGoalContract, CoopOwnershipShare
    └── seed.py              # seed rows for coop_beneficiary_patterns
```

## Naming convention

- **Class names:** `Coop` prefix — `CoopCooperative`, `CoopGoal`, `CoopGoalContract`, etc.
- **Table names:** `coop_` prefix — `coop_cooperatives`, `coop_goals`, `coop_goal_contracts`, etc.

This makes it unambiguous at a glance which tables belong to the new
engine vs. any legacy `circles`-domain tables still around mid-migration,
and avoids collisions if you keep both domains mounted side by side
during the cutover.

## Things you MUST adjust before this runs

1. **`mixins.py` → `from src.infra.db.base import Base`** — point this at
   your actual declarative Base / async session setup in
   `apps/backend/src/infra/db`.
2. **`users.id` foreign keys** (`created_by`, `beneficiary_user_id`,
   `CoopOwnershipShare.user_id`) — confirm the real table name for your
   existing `domains/users` model; adjusted here as a plain guess.
3. **Alembic migration** — nothing here is a migration yet, it's the
   model layer only. Once you're happy with the shape, run
   `alembic revision --autogenerate -m "add cooperative engine"` against
   this domain and review the generated DDL by hand (JSONB defaults and
   the `VersionedMixin` server_default on `effective_from` are the two
   things autogenerate tends to get slightly wrong).
4. **Cutover from `circles`** — this doesn't touch `domains/circles` or
   its existing data. You'll want a one-time data migration
   (`circles` → `coop_cooperatives`, existing group settings →
   `coop_governance_rules` / `coop_membership_rules` / `coop_financial_rules`
   rows at `version=1`) before deleting the old domain. Every place in
   `apps/backend/src/domains/{contributions,memberships,payments,...}`
   and `apps/frontend` that currently references `circle_id` will need to
   be repointed at `cooperative_id`.

## Deliberately NOT included here (see doc §7 "Next")

- Money layer: `contributions`, `wallets`, `payments`, `payouts` — these
  keep their existing domains, just repoint their FK from `circle_id` to
  `cooperative_id`.
- Trust/reputation layer: `trust_ledger_events`, `endorsements`,
  `leadership_roles`, `disputes` — not designed yet.
- Intelligence layer: `ai_insights`, `risk_flags` — not designed yet.
- Support infra: `notifications`, `analytics_snapshots` — existing
  domains, same repoint-the-FK treatment as money layer.

## Enforcing "one current-effective rule version"

`VersionedMixin` does NOT add a DB constraint for "only one row per
cooperative with `effective_to IS NULL`" — see the docstring in
`mixins.py` for why, and the partial-unique-index alternative if you'd
rather enforce it at the DB layer instead of the service layer.

import uuid

if TYPE_CHECKING:

    )
    from src.domains.cooperatives.models.goals import CoopGoal

    from src.domains.users.models import User

from src.domains.cooperatives.models.enums import CooperativeStatus, CooperativeModel

    Base,
    TimestampMixin,
    UUIDPrimaryKeyMixin,
)


class CoopCooperative(UUIDPrimaryKeyMixin, TimestampMixin, Base):
    """
    A Cooperative is a container that composes the four versioned rule 
    modules and owns zero or more Goals — it does NOT itself hold 
    financial/governance logic inline (that's the point of the engine 
    split).
    """

    __tablename__ = "coop_cooperatives"

    name: Mapped[str] = mapped_column(String(255), nullable=False)
    description: Mapped[str | None] = mapped_column(Text, nullable=True)
    cover_image: Mapped[str | None] = mapped_column(String(1024), nullable=True)

    status: Mapped[CooperativeStatus] = mapped_column(
        String(32), default=CooperativeStatus.PENDING, nullable=False
    )

    coop_model: Mapped[CooperativeModel] = mapped_column(
        String(32), default=CooperativeModel.COOP, nullable=False
    )

    created_by: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True), ForeignKey("users.id"), nullable=False
    )

    # --- relationships ---
    governance_rules: Mapped[list["CoopGovernanceRule"]] = relationship(
        back_populates="cooperative", cascade="all, delete-orphan"
    )
    membership_rules: Mapped[list["CoopMembershipRule"]] = relationship(
        back_populates="cooperative", cascade="all, delete-orphan"
    )
    financial_rules: Mapped[list["CoopFinancialRule"]] = relationship(
        back_populates="cooperative", cascade="all, delete-orphan"
    )
    decision_rules: Mapped[list["CoopDecisionRule"]] = relationship(
        back_populates="cooperative", cascade="all, delete-orphan"
    )
    goals: Mapped[list["CoopGoal"]] = relationship(
        back_populates="cooperative", cascade="all, delete-orphan"
    )
    memberships: Mapped[list["GroupMembership"]] = relationship(
        back_populates="group",
        cascade="all, delete-orphan",
        overlaps="users,memberships",
    )
    users: Mapped[list["User"]] = relationship(
        secondary="group_memberships",
        back_populates="cooperatives",
        viewonly=False,
        overlaps="group,memberships,user",
    )
    invitations: Mapped[list["CoopInvitation"]] = relationship(
        back_populates="cooperative",
        cascade="all, delete-orphan",
        overlaps="memberships,users",
    )

    activities: Mapped[list["CoopActivity"]] = relationship(
        back_populates="cooperative",
        cascade="all, delete-orphan",
    )

    def __repr__(self) -> str:
        return f"<CoopCooperative id={self.id} name={self.name!r} status={self.status}>"

"""
Enums for the cooperatives domain.

Kept as plain Python `str` Enums mapped via SQLAlchemy's `Enum(..., native_enum=False)`
so they store as VARCHAR + CHECK constraint rather than a Postgres native enum type.
Native Postgres enums are a pain to alter (ADD VALUE can't run in a transaction with
other DDL pre-PG12, and dropping/renaming a value requires a full type rebuild), and
this schema is expected to evolve — e.g. new `beneficiary_patterns.code` values,
new `decision_rules.mechanism` values. If you'd rather have native enums, flip
`native_enum=True` per-column later; the Python side doesn't change.
"""

import uuid
from datetime import datetime, date

from sqlalchemy import DateTime, Date, Integer, func
from sqlalchemy.dialects.postgresql import UUID
from sqlalchemy.orm import Mapped, mapped_column

from src.infra.db.database import Base 

__all__ = ["Base", "UUIDPrimaryKeyMixin", "TimestampMixin", "VersionedMixin"]


class UUIDPrimaryKeyMixin:
    id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True), primary_key=True, default=uuid.uuid4
    )


class TimestampMixin:
    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True), server_default=func.now(), nullable=False
    )
    updated_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        server_default=func.now(),
        onupdate=func.now(),
        nullable=False,
    )


class VersionedMixin:
    """
    Applied to the four rule-module tables (governance / membership /
    financial / decision). A cooperative's rules change over time, and
    disputes / AI governance reasoning / audit history must resolve
    against the rules as they stood *at the time*, not today's — so every
    edit is a new row, never an UPDATE of an existing one.

    `effective_to IS NULL` means "currently in effect".
    "at most one currently-effective row per (cooperative_id[, decision_type])"
    enforced at the application/service layer (e.g. in the same transaction that
    inserts a new version, the previous row's `effective_to` is closed out)
    rather than as a DB constraint, since Postgres partial unique indexes
    on NULL don't compose cleanly with the extra `decision_type` grouping
    key on `decision_rules`.
    """

    version: Mapped[int] = mapped_column(Integer, nullable=False, default=1)
    effective_from: Mapped[date] = mapped_column(
        Date, nullable=False, server_default=func.current_date()
    )
    effective_to: Mapped[date | None] = mapped_column(Date, nullable=True)

from __future__ import annotations

from typing import Generic, Sequence, TypeVar
from uuid import UUID

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

T = TypeVar("T")


class RulesRepository(Generic[T]):
    def __init__(self, session: AsyncSession) -> None:
        self._session = session

    async def add(self, instance: T) -> T:
        self._session.add(instance)
        await self._session.flush()
        return instance

    async def get_by_id(self, model_cls: type[T], rule_id: UUID | str) -> T | None:
        rule_id = str(rule_id)
        result = await self._session.execute(select(model_cls).where(model_cls.id == rule_id))
        return result.scalar_one_or_none()

    async def get_active_for_cooperative(self, model_cls: type[T], cooperative_id: UUID | str) -> T | None:
        cooperative_id = str(cooperative_id)
        result = await self._session.execute(
            select(model_cls).where(model_cls.cooperative_id == cooperative_id).order_by(model_cls.version.desc())
        )
        return result.scalar_one_or_none()

    async def list_versions(self, model_cls: type[T], cooperative_id: UUID | str) -> Sequence[T]:
        cooperative_id = str(cooperative_id)
        result = await self._session.execute(
            select(model_cls)
            .where(model_cls.cooperative_id == cooperative_id)
            .order_by(model_cls.version.desc())
        )
        return result.scalars().all()

@router.post("/create", response_model=CoopGroupDetails, status_code=status.HTTP_201_CREATED)
async def create_cooperative_group(
    coop_data: CoopGroupCreate,
    user: AuthenticatedUser = Depends(get_current_user),
    db: AsyncSession = Depends(get_async_db_session),
):
    # 1. Resolve phone numbers → Flow addresses
    # TODO: real lookup once user phone is indexed
    # member_addresses = await resolve_member_addresses(db, coop_data.member_phones)
    member_addresses = [user.flow_address] if user.flow_address else []

    # 2. Convert local currency → USDC
    usdc_amount = await fx_service.to_usdc(
        coop_data.contribution_amount, coop_data.currency
    )

    print(f"\nAmount in USDC: {usdc_amount} \n")

    # 3. Write to Postgres first with status=pending, chain_circle_id=None
    #    So a Flow failure doesn't lose the user's data
    pending_data = coop_data.model_copy(update={
        "creator_id": user.id,
        "chain_circle_id": None,        # filled in after Flow confirms
        "weekly_amount_usdc": usdc_amount,
        "status": CooperativeStatus.pending.value,
        "current_round": 0,
        "is_complete": False,
    })
    coop = await CooperativeGroupService.create_coop(pending_data, db)

    # 4. Submit to Flow blockchain
    try:
        tx_id = await flow_service.create_circle(
            member_addresses=member_addresses,
            weekly_amount_usdc=usdc_amount,
            rotation_order=coop_data.rotation_order,
        )
        chain_circle_id = await flow_service.await_cicle_created_event(tx_id)

        print(f"\n tx_id: {tx_id} \n chain_circle_id: {chain_circle_id} \n")

        # Update record with chain data
        coop.chain_circle_id = chain_circle_id
        coop.status = CooperativeStatus.active.value
        await db.commit()
        await db.refresh(coop)
    except Exception as e:
        logger.error(f"Flow tx failed for circle {coop.id}: {e}")
        # Circle stays in DB with status=pending — can be retried
        # Don't raise — return the pending circle to the user

    # 5. Add creator membership
    membership_data = MembershipCreate(
        user_id=user.id,
        group_id=coop.id,
        invited_by=user.id,
        role="admin",
        status="accepted",
        payout_position=1 if coop_data.rotation_order == "sequential" else None,
    )
    await CooperativeMembershipService.create_membership(db, membership_data, user)

    # 6. Activity log
    activity_data = ActivityCreate(
        user_id=coop.creator_id,
        type=ActivityType.created_group.value,
        description=f"You created a group: {coop.name}",
        group_id=coop.id,
        entity_id=str(coop.id),
        amount=None,
    )
    await ActivityService.log(db, activity_data)

    # 7. Notification
    try:
        noti_data = NotificationCreate(
            user_id=user.id,
            title="New Circle Created",
            message=f"Your circle '{coop.name}' was successfully created on Flow.",
            event_type="group",
            type="success",
            entity_url=f"/circle/{coop.id}",
        )
        await NotificationService.create_and_push_notification_to_user(noti_data, db)
    except Exception as e:
        logger.warning(f"Notification failed (non-fatal): {e}")

    # TODO: Send invite notifications to coop_data.member_phones

    return CoopGroupDetails.model_validate(coop)

@router.post("/{circle_id}/invite")
async def generate_circle_invite(
    circle_id: str,
    user: AuthenticatedUser = Depends(get_current_user),
    db: AsyncSession = Depends(get_async_db_session),
):
    # 1. Confirm circle exists
    circle = await CooperativeGroupService.get_coop_group_by_id(db, circle_id)
    if not circle:
        raise HTTPException(status_code=404, detail="Circle not found")

    # 2. Confirm caller is an accepted member
    membership = await CooperativeMembershipService.get_membership_by_user_and_group(
        user_id=user.id, group_id=UUID(circle_id), db=db
    )

    if not membership or membership.status.value != "accepted":
        raise HTTPException(
            status_code=403,
            detail="Only accepted members can generate invite links",
        )

    # 3. Generate the code
    invite_code = await CooperativeMembershipService.generate_invite_code(
        db=db,
        group_id=UUID(circle_id),
        inviter_id=user.id,
    )

    base_url = AppConfig.CLIENT_DOMAIN  # e.g. "https://coopwise.app" or "http://localhost:3000"
    invite_link = f"{base_url}/invite/{invite_code}"

    return {
        "invite_code": invite_code,
        "invite_link": invite_link,
    }

@router.post("/{coop_id}/join", response_model=CoopGroupDetails)
async def join_circle(
    coop_id: str,
    user: AuthenticatedUser = Depends(get_current_user),
    db: AsyncSession = Depends(get_async_db_session),
):
    """
    Authenticated. Adds user as a member, calls Flow stub, returns updated circle.
    """
    coop = await CooperativeGroupService.get_coop_group_by_id(db, coop_id)
    if not coop:
        raise HTTPException(status_code=404, detail="Circle not found")

    if coop.status == CooperativeStatus.inactive:
        raise HTTPException(status_code=400, detail="This circle is no longer active")

    # Check if already a member
    existing = await CooperativeMembershipService.get_membership_by_user_and_group(
        user.id, UUID(coop_id), db
    )
    if existing and existing.status == "accepted":
        raise HTTPException(status_code=400, detail="Already a member of this circle")

    # 3. Check capacity
    if coop.member_count >= coop.max_members:
        raise HTTPException(status_code=400, detail="This circle is full")

    # Get next queue position
    count_stmt = select(func.count()).where(
        GroupMembership.group_id == UUID(coop_id),
        GroupMembership.status == "accepted",
    )
    
    member_count = (await db.execute(count_stmt)).scalar() or 0
    next_position = member_count + 1

    
    # Submit to Flow stub — replace with real JoinCircle.cdc call later
    try:
        tx_id = await flow_service.join_circle(
            circle_id=coop.chain_circle_id or 0,
            member_address=user.flow_address or "",
        )
    except Exception as e:
        logger.warning(f"Flow join_circle stub failed (non-fatal): {e}")
        tx_id = None
    
    print("\n COnfirmed Memberships - next count", member_count, next_position, "\n")
    # Create membership
    membership_data = MembershipCreate(
        user_id=user.id,
        group_id=UUID(coop_id),
        invited_by=coop.creator_id,
        role="member",
        status="accepted",
        queue_position=next_position,
    )
    await CooperativeMembershipService.create_membership(db, membership_data, user)

    print("\n Created membership\n")


    # Activity log
    try:
        await ActivityService.log(db, ActivityCreate(
            user_id=user.id,
            type=ActivityType.joined_group.value,
            description=f"You joined circle: {coop.name}",
            group_id=UUID(coop_id),
            entity_id=coop_id,
            amount=None,
        ))
    except Exception as e:
        logger.warning(f"Activity log failed (non-fatal): {e}")

    # Notification
    try:
        await NotificationService.create_and_push_notification_to_user(
            NotificationCreate(
                user_id=user.id,
                title="Circle joined",
                message=f"You are now a member of '{coop.name}'.",
                event_type="group",
                type="success",
                entity_url=f"/dashboard/circle/{coop_id}",
            ),
            db,
        )
    except Exception as e:
        logger.warning(f"Notification failed (non-fatal): {e}")

    updated = await CooperativeGroupService.get_coop_group_by_id(
        db, coop_id, requesting_user_id=str(user.id)
    )
    return updated

class CooperativeGroupService:
    # Create and manage cooperative groups.

    @staticmethod
    async def create_coop(coop_data: CoopGroupCreate, db: AsyncSession) -> CoopCooperative:
        try:
            new_coop_group = CoopCooperative(
                name=coop_data.name,
                description=coop_data.description,

                creator_id=coop_data.creator_id,
                max_members=coop_data.max_members,

                detail=f"Could not create cooperative group — {str(e)}",
            )

    @staticmethod
    async def get_coop_groups(db: AsyncSession, skip: int = 0, limit: int = 10):
        try:
            stmt = (
                select(CoopCooperative)
                .order_by(desc(CoopCooperative.created_at))
                .offset(skip)
                .limit(limit)
            )
            result = await db.execute(stmt)
            coop_groups = result.scalars().all()
        except Exception as e:
            logger.error(e)
            raise e
        return coop_groups








    @staticmethod
    async def get_coop_group_by_id(
        db: AsyncSession, coop_group_id: str, requesting_user_id: Optional[str] = None
    ) -> Optional[CoopGroupDetails]:
        try:
            stmt = select(CoopCooperative).where(
                CoopCooperative.id == coop_group_id
            )
            result = await db.execute(stmt)
            coop = result.scalars().first()
            if not coop:
                return None

            # Member count
            count_stmt = select(func.count()).where(
                GroupMembership.group_id == coop.id,
                GroupMembership.status == "accepted",
            )
            member_count = (await db.execute(count_stmt)).scalar() or 0

            details = CoopGroupDetails.model_validate(coop)
            details.member_count = member_count

            # Caller's queue position
            if requesting_user_id:
                pos_stmt = select(GroupMembership.payout_position).where(
                    GroupMembership.group_id == coop.id,
                    GroupMembership.user_id == requesting_user_id,
                )
                pos = (await db.execute(pos_stmt)).scalar()
                details.your_position_in_queue = pos

            return details
        except Exception as e:
            logger.error(f"get_coop_group_by_id: {e}")
            raise

    @staticmethod
    async def get_ext_coop_group_by_id(db: AsyncSession, coop_group_id: str):
        # 1. Fetch group
        stmt = select(CoopCooperative).where(CoopCooperative.id == coop_group_id)
        result = await db.execute(stmt)
        group: CoopCooperative = result.scalars().first()

        if not group:
            return None

        # 2. Get active memberships (ACCEPTED)
        membership_stmt = (
            select(GroupMembership)
            .where(
                GroupMembership.group_id == coop_group_id,
                GroupMembership.status == MembershipStatus.accepted,
            )
            .options(joinedload(GroupMembership.user))
        )
        members_result = await db.execute(membership_stmt)
        members = members_result.scalars().all()
        member_ids = [m.user_id for m in members]

        # 3. Get contributions by accepted members (completed)
        contrib_stmt = select(Contribution).where(
            Contribution.group_id == coop_group_id,
            Contribution.user_id.in_(member_ids),
            Contribution.status == ContributionStatus.completed,
        )
        contrib_result = await db.execute(contrib_stmt)
        contributions = contrib_result.scalars().all()

        # Calculate total_saved from the contributions you just fetched
        total_saved = sum(c.amount or 0 for c in contributions)

        # 4. Compute progress
        progress = (
            float(total_saved) / float(group.target_amount) * 100
            if group.target_amount
            else 0
        )

        # 5. Determine next contribution date
        today = datetime.datetime.now()
        freq = group.contribution_frequency
        next_contrib_date = None
        if freq == ContributionFrequency.weekly:
            next_contrib_date = today + datetime.timedelta(weeks=1)
        elif freq == ContributionFrequency.monthly:
            next_contrib_date = today + datetime.timedelta(days=30)
        elif freq == ContributionFrequency.daily:
            next_contrib_date = today + datetime.timedelta(days=1)

        next_contribution = {
            "amount": float(group.contribution_amount),
            "due_date": next_contrib_date.isoformat(),
            "days_left": (
                (next_contrib_date - today).days if next_contrib_date else None
            ),
        }

        # 6. Determine next payout
        next_payout = {
            "amount": float(group.contribution_amount) * len(member_ids),
            "recipient": "TBD",
            "date": (
                group.next_payout_date.isoformat() if group.next_payout_date else None
            ),
        }

        # 7. Build final response
        return {
            **group.__dict__,
            "total_saved": float(total_saved),
            "progress": round(progress, 2),
            "members": [MembershipExtDetails.model_validate(m) for m in members],
            "members_count": len(member_ids),
            "contributions": [
                ContributionDetail.model_validate(c) for c in contributions
            ],
            "next_contribution": next_contribution,
            "next_payout": next_payout,
        }

    @staticmethod
    async def get_user_and_suggested_groups(
        user: AuthenticatedUser,
        db: AsyncSession,
        redis: Redis,
        page: int = 1,
        page_size: int = 10,
    ) -> ExploreGroups:
        cache_key = f"groups:{user.id}:page:{page}"
        cached = await get_cache(cache_key)
        if cached:
            logger.info(
                f"🔄 Using cached user & suggested groups for user {user.id} (page {page})"
            )
            return cached

        logger.info(
            f"Fetching user & suggested groups for user {user.id} (page {page})"
        )

        # 1. Fetch user's group memberships
        user_group_stmt = (
            select(CoopCooperative)
            .join(GroupMembership, CoopCooperative.id == GroupMembership.group_id)
            .where(GroupMembership.user_id == user.id)
            .options(joinedload(CoopCooperative.memberships))
        )
        user_groups_result = await db.execute(user_group_stmt)
        user_groups: List[CoopCooperative] = (
            user_groups_result.unique().scalars().all()
        )
        user_group_ids = [group.id for group in user_groups]

        # 2. Fetch suggested groups (not already joined)
        offset = (page - 1) * page_size
        suggested_group_stmt = (
            select(CoopCooperative)
            .where(
                CoopCooperative.status == CooperativeStatus.active,
                CoopCooperative.id.notin_(user_group_ids),
            )
            .limit(page_size)
            .offset(offset)
        )
        suggested_groups_result = await db.execute(suggested_group_stmt)
        suggested_groups: List[CoopCooperative] = (
            suggested_groups_result.scalars().all()
        )

        result = ExploreGroups(
            user_groups=[CoopGroupDetails.model_validate(g) for g in user_groups],
            suggested_groups=[
                CoopGroupDetails.model_validate(g) for g in suggested_groups
            ],
        )
        result_serializable = jsonable_encoder(result)

        await update_cache(cache_key, result_serializable, ttl=300)
        logger.info(
            f"📦 Cached user & suggested groups for user {user.id} (page {page})"
        )

        return result_serializable

        created_by_uuid = created_by if isinstance(created_by, UUID) else UUID(str(created_by))
        cooperative = CoopCooperative(

            created_by=created_by_uuid,
        )
        await self._uow.cooperatives.add(cooperative)
        await self._uow.commit()
        return cooperative

        return await self._uow.cooperatives.list(limit=page_size)

        cooperative = await self._uow.cooperatives.get_by_id(cooperative_id)
        if cooperative is None:
            raise CooperativeNotFoundError(str(cooperative_id))
        return cooperative

    async def get_cooperative_detail(self, cooperative_id: UUID | str) -> dict[str, Any]:
        cooperative = await self.get_cooperative(cooperative_id)
        return {
            "cooperative": cooperative,
            "rules": {},
            "member_count": 0,
            "goal_count": 0,
        }

    async def update(self, cooperative_id: UUID | str, acting_user_id: UUID | str, **fields: Any) -> CoopCooperative:
        cooperative = await self.get_cooperative(cooperative_id)
        if acting_user_id is not None:
            # auth is intentionally lightweight at this scaffold stage
            pass
        for key, value in fields.items():
            if key in {"name", "description", "cover_image"}:
                setattr(cooperative, key, value)
        await self._uow.commit()
        return cooperative

    async def activate(self, cooperative_id: UUID | str, acting_user_id: UUID | str) -> CoopCooperative:
        cooperative = await self.get_cooperative(cooperative_id)
        if cooperative.status != CooperativeStatus.PENDING:
            raise IllegalStatusTransitionError(f"{cooperative.status} -> {CooperativeStatus.ACTIVE}")
        if acting_user_id is None:
            raise NotAuthorizedError()

        cooperative.status = CooperativeStatus.ACTIVE
        await self._uow.commit()
        if self._logger is not None:
            self._logger.info(
                "cooperative.activated",
                extra={"cooperative_id": str(cooperative.id), "actor_id": str(acting_user_id)},
            )
        await self._events.dispatch(CooperativeActivated(cooperative_id=cooperative.id, actor_id=UUID(str(acting_user_id))))
        return cooperative

from __future__ import annotations

from dataclasses import dataclass

from sqlalchemy.ext.asyncio import AsyncSession

from src.domains.cooperatives.repositories.beneficiary_pattern_repository import BeneficiaryPatternRepository

from src.domains.cooperatives.repositories.ownership_share_repository import OwnershipShareRepository
from src.domains.cooperatives.repositories.rules_repository import RulesRepository


class UnitOfWork:
    def __init__(self, session: AsyncSession):
        self.session = session
        self.cooperatives = CooperativeRepository(session)
        self.rules = RulesRepository(session)          # generic, shared across the 4 rule types
        self.goals = GoalRepository(session)
        self.ownership_shares = OwnershipShareRepository(session)

    async def __aenter__(self) -> "UnitOfWork":
        return self

    async def __aexit__(self, exc_type, exc, tb) -> None:
        if exc_type is not None:
            await self.session.rollback()
        # no implicit commit on clean exit — commit is explicit, see below

    async def commit(self) -> None:
        await self.session.commit()

    async def rollback(self) -> None:
        await self.session.rollback()
